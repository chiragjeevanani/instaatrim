<script data-purpose="auth-flow-logic">
    const phoneView = document.getElementById('phone-view');
    const otpView = document.getElementById('otp-view');
    const backBtn = document.getElementById('btn-back');
    const skipBtn = document.getElementById('btn-skip');
    const phoneInput = document.getElementById('phone-input');
    const continueBtn = document.getElementById('btn-continue');
    const displayPhone = document.getElementById('display-phone');
    const otpBoxes = document.querySelectorAll('.otp-box');
    const timerElement = document.getElementById('timer');
    
    let timerInterval = null;
    let secondsLeft = 13;

    // Handle Phone Input & Enable Continue Button
    function handlePhoneInput(input) {
      // Allow only numbers
      input.value = input.value.replace(/[^0-9]/g, '');
      
      if (input.value.length === 10) {
        continueBtn.disabled = false;
        continueBtn.classList.remove('bg-gray-400/80', 'cursor-not-allowed');
        continueBtn.classList.add('bg-[#891845]', 'hover:bg-[#721339]', 'cursor-pointer');
      } else {
        continueBtn.disabled = true;
        continueBtn.classList.add('bg-gray-400/80', 'cursor-not-allowed');
        continueBtn.classList.remove('bg-[#891845]', 'hover:bg-[#721339]', 'cursor-pointer');
      }
    }

    // Auto-fill from Truecaller
    function fillTruecaller() {
      phoneInput.value = '7000792773';
      handlePhoneInput(phoneInput);
      switchToOTPScreen();
    }

    // Transition to OTP verification
    function switchToOTPScreen() {
      const currentNumber = phoneInput.value || '7000792773';
      displayPhone.textContent = currentNumber;
      
      phoneView.classList.add('hidden');
      otpView.classList.remove('hidden');
      
      // Top bar changes
      backBtn.classList.remove('opacity-0', 'pointer-events-none');
      skipBtn.classList.add('hidden');

      // Start OTP Timer
      startCountdown();

      // Focus first input box
      setTimeout(() => {
        otpBoxes[0].focus();
      }, 100);
    }

    // Transition back to Phone input
    function switchToPhoneScreen() {
      otpView.classList.add('hidden');
      phoneView.classList.remove('hidden');
      
      backBtn.classList.add('opacity-0', 'pointer-events-none');
      skipBtn.classList.remove('hidden');

      if (timerInterval) clearInterval(timerInterval);
    }

    // Quick debug toggle by tapping brand logo
    function toggleScreen() {
      if (otpView.classList.contains('hidden')) {
        switchToOTPScreen();
      } else {
        switchToPhoneScreen();
      }
    }

    // Countdown Timer logic
    function startCountdown() {
      if (timerInterval) clearInterval(timerInterval);
      secondsLeft = 13;
      timerElement.textContent = `${secondsLeft}s`;
      
      timerInterval = setInterval(() => {
        secondsLeft--;
        if (secondsLeft <= 0) {
          clearInterval(timerInterval);
          timerElement.textContent = '0s';
        } else {
          timerElement.textContent = `${secondsLeft}s`;
        }
      }, 1000);
    }

    // Handle auto-focus jump for 4 OTP digit boxes
    otpBoxes.forEach((box, index) => {
      box.addEventListener('input', (e) => {
        if (box.value.length >= 1) {
          box.value = box.value.slice(0, 1);
          if (index < otpBoxes.length - 1) {
            otpBoxes[index + 1].focus();
          }
        }
      });

      box.addEventListener('keydown', (e) => {
        if (e.key === 'Backspace' && box.value === '' && index > 0) {
          otpBoxes[index - 1].focus();
        }
      });
    });
  </script>
